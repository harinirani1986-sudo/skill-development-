<body ng-controller="myCtrl">

    <div class="container">

        <h2>Student Application Form</h2>

        <form name="myForm">

            <label>Name</label>
            <input type="text"
                   name="username"
                   ng-model="user.name"
                   required>

            <span class="error"
                  ng-show="myForm.username.$touched && myForm.username.$invalid">
                  Name is required
            </span>

            
            <label>Email</label>
            <input type="email"
                   name="email"
                   ng-model="user.email"
                   required>

            <span class="error"
                  ng-show="myForm.email.$error.email">
                  Invalid Email
            </span>
 
            <label>Age</label>
            <input type="number"
                   ng-model="user.age">

            
            <button ng-click="submitForm()"
                    ng-disabled="myForm.$invalid">
                    Submit
            </button>

        </form>

        
        <div class="output" ng-if="submitted">

            <h3>Submitted Details</h3>

            
            <p><b>Name:</b> {{ user.name | uppercase }}</p>

            <p><b>Email:</b> {{ user.email | lowercase }}</p>

            <p><b>Message:</b> {{ message }}</p>

        </div>

    </div>

    <script>

        
        var app = angular.module("myApp", []);

        
        app.service("myService", function() {

            this.successMessage = function() {
                return "Application Submitted Successfully";
            };

        });

       
        app.controller("myCtrl", function($scope, myService) {

            $scope.submitted = false;

            $scope.submitForm = function() {

                $scope.message = myService.successMessage();

                $scope.submitted = true;
            };

        });

    </script>
